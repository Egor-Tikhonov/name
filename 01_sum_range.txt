int main() {
int a,b,s=0;

#include <iostream>
using namespace std;

int main() {
    long long a, b;
    cin>>a>>b;

    if (a > b) {
        long long temp = a;
        a = b;
        b = temp;
    }
    long long count = b - a + 1;
    long long s = (a + b) * count / 2;

    cout << "sum=" << s << endl;
    return 0;
}
