#include <iostream>
#include <limits>

int main() {
    long long from, to;

    std::cout << "Enter range [from, to]: ";
    if (!(std::cin >> from >> to)) {
        std::cerr << "Error: invalid input.\n";
        return 1;
    }

    long long sum = 0;
    if (from <= to) {
        for (long long i = from; i <= to; ++i) {
            sum += i;
        }
    } else {
        std::cout << "Sum: 0 (range is empty)\n";
        return 0;
    }

    std::cout << "Sum: " << sum << '\n';
    return 0;
}
